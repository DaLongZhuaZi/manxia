# 图源详情页搜索增强方案

> **版本**: v1.0  
> **日期**: 2026-01-09  
> **状态**: ✅ 已实施

---

## 一、目标概述

### 1.1 当前问题

目前图源详情页搜索功能仅支持**关键词搜索**，用户输入的内容直接作为 `keyword` 传递给 `workflows.search`。无法实现：

- **ID直达**：通过作品ID、作者ID、系列ID直接定位内容
- **URL直达**：粘贴完整URL自动解析并跳转
- **智能识别**：自动区分纯数字是作品ID还是作者ID

### 1.2 增强目标

实现**全图源通用**的搜索增强，支持：

| 输入类型 | 示例 | 预期行为 |
|---------|------|---------|
| 普通关键词 | `海贼王` | 执行 `workflows.search` |
| URL | `https://pixiv.net/artworks/12345` | 解析URL，直达详情页 |
| 作品ID前缀 | `id:12345` 或 `aid:12345` | 调用 `getMangaDetail` 或 `searchById` |
| 作者ID前缀 | `uid:67890` 或 `user:67890` | 调用 `searchByAuthor`（如有）或降级搜索 |
| 系列ID前缀 | `sid:111` 或 `series:111` | 调用 `searchBySeries`（如有）或降级搜索 |
| 纯数字 | `12345` | 弹窗询问：作品ID / 作者ID / 关键词搜索 |

```typescript
import { logger } from '../../Utils/Logger';
import { 
  SearchIntent, 
  SearchIntentType, 
  UrlParsedInfo,
  SourceUrlPattern 
} from './SearchIntentTypes';

const TAG = 'SearchIntentParser';

/**
 * 搜索意图解析器
 * 将用户输入解析为结构化的搜索意图
 */
export class SearchIntentParser {
  
  /** ID前缀映射 */
  private static readonly ID_PREFIXES: Map<string, SearchIntentType> = new Map([
    ['id:', SearchIntentType.MANGA_ID],
    ['aid:', SearchIntentType.MANGA_ID],
    ['manga:', SearchIntentType.MANGA_ID],
    ['illust:', SearchIntentType.MANGA_ID],
    ['uid:', SearchIntentType.AUTHOR_ID],
    ['user:', SearchIntentType.AUTHOR_ID],
    ['author:', SearchIntentType.AUTHOR_ID],
    ['sid:', SearchIntentType.SERIES_ID],
    ['series:', SearchIntentType.SERIES_ID],
    ['cid:', SearchIntentType.CHAPTER_ID],
    ['chapter:', SearchIntentType.CHAPTER_ID],
  ]);

  /**
   * 解析用户输入的搜索查询
   * @param rawQuery 原始输入
   * @param sourceUrlPattern 图源URL模式配置（可选）
   * @returns 解析后的搜索意图
   */
  static parse(rawQuery: string, sourceUrlPattern?: SourceUrlPattern): SearchIntent {
    const trimmed = rawQuery.trim();
    
    if (!trimmed) {
      return {
        type: SearchIntentType.KEYWORD,
        rawQuery: rawQuery
      };
    }

    // 1. 优先检测URL
    if (this.isUrl(trimmed)) {
      return this.parseUrl(trimmed, sourceUrlPattern);
    }

    // 2. 检测显式ID前缀
    const prefixResult = this.parsePrefixedId(trimmed);
    if (prefixResult) {
      return prefixResult;
    }

    // 3. 检测纯数字（需要用户选择）
    if (this.isPureNumber(trimmed)) {
      return {
        type: SearchIntentType.AMBIGUOUS,
        id: trimmed,
        rawQuery: rawQuery
      };
    }

    // 4. 默认为关键词搜索
    return {
      type: SearchIntentType.KEYWORD,
      rawQuery: rawQuery
    };
  }

  /**
   * 检测是否为URL
   */
  private static isUrl(text: string): boolean {
    return text.startsWith('http://') || text.startsWith('https://');
  }

  /**
   * 解析URL
   */
  private static parseUrl(url: string, sourceUrlPattern?: SourceUrlPattern): SearchIntent {
    logger.info(TAG, `解析URL: ${url}`);
    
    try {
      const urlObj = new URL(url);
      const domain = urlObj.hostname;
      const path = urlObj.pathname;

      // 通用Pixiv URL解析
      if (domain.includes('pixiv')) {
        return this.parsePixivUrl(url, path);
      }

      // 使用图源自定义模式解析
      if (sourceUrlPattern) {
        const customResult = this.parseWithCustomPattern(url, path, sourceUrlPattern);
        if (customResult) {
          return customResult;
        }
      }

      // 无法解析，返回URL类型让图源自行处理
      return {
        type: SearchIntentType.URL,
        url: url,
        rawQuery: url,
        urlParsedInfo: {
          domain: domain,
          pathType: 'unknown'
        }
      };
    } catch (e) {
      logger.warn(TAG, `URL解析失败: ${e}`);
      return {
        type: SearchIntentType.KEYWORD,
        rawQuery: url
      };
    }
  }

  /**
   * 解析Pixiv URL
   */
  private static parsePixivUrl(url: string, path: string): SearchIntent {
    // /artworks/12345 或 /i/12345
    const artworkMatch = path.match(/\/(artworks|i)\/(\d+)/);
    if (artworkMatch) {
      return {
        type: SearchIntentType.MANGA_ID,
        id: artworkMatch[2],
        url: url,
        rawQuery: url,
        urlParsedInfo: {
          domain: 'pixiv.net',
          pathType: 'artwork',
          extractedId: artworkMatch[2]
        }
      };
    }

    // /users/67890 或 /member.php?id=67890
    const userMatch = path.match(/\/users\/(\d+)/) || url.match(/member\.php\?id=(\d+)/);
    if (userMatch) {
      return {
        type: SearchIntentType.AUTHOR_ID,
        id: userMatch[1],
        url: url,
        rawQuery: url,
        urlParsedInfo: {
          domain: 'pixiv.net',
          pathType: 'user',
          extractedId: userMatch[1]
        }
      };
    }

    // /user/67890/series/111
    const seriesMatch = path.match(/\/user\/(\d+)\/series\/(\d+)/);
    if (seriesMatch) {
      return {
        type: SearchIntentType.SERIES_ID,
        id: `${seriesMatch[1]}-${seriesMatch[2]}`, // userId-seriesId
        url: url,
        rawQuery: url,
        urlParsedInfo: {
          domain: 'pixiv.net',
          pathType: 'series',
          extractedId: seriesMatch[2]
        }
      };
    }

    return {
      type: SearchIntentType.URL,
      url: url,
      rawQuery: url,
      urlParsedInfo: {
        domain: 'pixiv.net',
        pathType: 'unknown'
      }
    };
  }

  /**
   * 使用自定义模式解析URL
   */
  private static parseWithCustomPattern(
    url: string, 
    path: string, 
    pattern: SourceUrlPattern
  ): SearchIntent | null {
    if (pattern.artworkPattern) {
      const regex = new RegExp(pattern.artworkPattern);
      const match = path.match(regex);
      if (match && match[1]) {
        return {
          type: SearchIntentType.MANGA_ID,
          id: match[1],
          url: url,
          rawQuery: url
        };
      }
    }

    if (pattern.authorPattern) {
      const regex = new RegExp(pattern.authorPattern);
      const match = path.match(regex);
      if (match && match[1]) {
        return {
          type: SearchIntentType.AUTHOR_ID,
          id: match[1],
          url: url,
          rawQuery: url
        };
      }
    }

    return null;
  }

  /**
   * 解析带前缀的ID
   */
  private static parsePrefixedId(text: string): SearchIntent | null {
    const lowerText = text.toLowerCase();
    
    for (const [prefix, intentType] of this.ID_PREFIXES) {
      if (lowerText.startsWith(prefix)) {
        const id = text.substring(prefix.length).trim();
        if (id) {
          return {
            type: intentType,
            id: id,
            rawQuery: text
          };
        }
      }
    }
    
    return null;
  }

  /**
   * 检测是否为纯数字
   */
  private static isPureNumber(text: string): boolean {
    return /^\d+$/.test(text);
  }
}
```

```typescript
/**
 * 通过ID搜索/获取单个作品
 * @param illustId 作品ID
 * @returns 搜索结果（单个作品包装为列表）
 */
async searchById(illustId: string): Promise<EngineResult<SearchResult>> {
  if (!this.config) {
    throw new MangaSourceError(MangaSourceErrorCode.CONFIG_NOT_LOADED, '配置未加载');
  }

  const startTime = Date.now();
  logger.info(TAG, `开始ID直达搜索: illustId="${illustId}"`);

  try {
    // 检查是否有专用工作流
    const hasSearchById = this.configParser.hasWorkflow(this.config, 'searchById');
    
    let actions: Action[];
    if (hasSearchById) {
      actions = this.configParser.buildSearchByIdActions(this.config, illustId);
    } else {
      // 降级：使用 getMangaDetail 获取详情，包装为搜索结果
      logger.info(TAG, '图源未实现searchById，使用getMangaDetail降级');
      const detailResult = await this.getMangaDetail(illustId);
      if (detailResult.success && detailResult.data) {
        return {
          success: true,
          data: {
            mangas: [detailResult.data.detail],
            total: 1,
            page: 1,
            hasNext: false
          },
          duration: Date.now() - startTime,
          actionCount: 1
        };
      }
      return {
        success: false,
        error: detailResult.error || 'ID搜索失败',
        duration: Date.now() - startTime,
        actionCount: 0
      };
    }

    // 执行工作流
    const context = this.createActionContext();
    await this.populateSourceSettings(context);
    context.variables['illustId'] = illustId;
    context.variables['mangaId'] = illustId;

    let results: MangaInfo[] = [];
    for (const action of actions) {
      const result = await this.actionEngine.executeAction(action, context);
      if (!result.success && !action.optional) {
        throw new MangaSourceError(MangaSourceErrorCode.ACTION_FAILED, result.error || '操作失败');
      }
      if ((action.type === 'extract' || action.type === 'api' || action.type === 'script') && result.data) {
        results = this.extractMangaResults(result.data);
      }
    }

    return {
      success: true,
      data: {
        mangas: results,
        total: results.length,
        page: 1,
        hasNext: false
      },
      duration: Date.now() - startTime,
      actionCount: actions.length
    };
  } catch (error) {
    const errorMessage = error instanceof Error ? error.message : String(error);
    logger.error(TAG, `ID搜索失败: ${errorMessage}`);
    return {
      success: false,
      error: errorMessage,
      duration: Date.now() - startTime,
      actionCount: 0
    };
  }
}

/**
 * 通过作者ID搜索作品列表
 * @param authorId 作者ID
 * @param page 页码
 * @returns 搜索结果
 */
async searchByAuthor(authorId: string, page: number = 1): Promise<EngineResult<SearchResult>> {
  if (!this.config) {
    throw new MangaSourceError(MangaSourceErrorCode.CONFIG_NOT_LOADED, '配置未加载');
  }

  const startTime = Date.now();
  logger.info(TAG, `开始作者搜索: authorId="${authorId}", page=${page}`);

  // 检查是否支持作者搜索
  const hasSearchByAuthor = this.configParser.hasWorkflow(this.config, 'searchByAuthor');
  if (!hasSearchByAuthor) {
    logger.warn(TAG, '图源未实现searchByAuthor工作流');
    return {
      success: false,
      error: '该图源不支持作者ID搜索',
      duration: Date.now() - startTime,
      actionCount: 0
    };
  }

  try {
    const actions = this.configParser.buildSearchByAuthorActions(this.config, authorId);
    const context = this.createActionContext();
    await this.populateSourceSettings(context);
    context.variables['authorId'] = authorId;
    context.variables['userId'] = authorId;
    context.variables['page'] = page.toString();

    let results: MangaInfo[] = [];
    for (const action of actions) {
      const result = await this.actionEngine.executeAction(action, context);
      if (!result.success && !action.optional) {
        throw new MangaSourceError(MangaSourceErrorCode.ACTION_FAILED, result.error || '操作失败');
      }
      if ((action.type === 'extract' || action.type === 'api' || action.type === 'script') && result.data) {
        results = this.extractMangaResults(result.data);
      }
    }

    return {
      success: true,
      data: {
        mangas: results,
        total: results.length,
        page: page,
        hasNext: results.length >= 20
      },
      duration: Date.now() - startTime,
      actionCount: actions.length
    };
  } catch (error) {
    const errorMessage = error instanceof Error ? error.message : String(error);
    logger.error(TAG, `作者搜索失败: ${errorMessage}`);
    return {
      success: false,
      error: errorMessage,
      duration: Date.now() - startTime,
      actionCount: 0
    };
  }
}

/**
 * 检查图源是否支持指定能力
 */
supportsCapability(capability: string): boolean {
  if (!this.config) return false;
  return this.configParser.hasWorkflow(this.config, capability);
}

/**
 * 从结果数据中提取漫画列表（辅助方法）
 */
private extractMangaResults(data: ESObject): MangaInfo[] {
  // 复用现有的结果提取逻辑
  if (Array.isArray(data)) {
    return data.map((item: Record<string, Object>) => this.transformToMangaInfo(item));
  }
  if (typeof data === 'string') {
    try {
      let parsed = data;
      if (parsed.startsWith('"') && parsed.endsWith('"')) {
        parsed = JSON.parse(parsed);
      }
      const jsonData = JSON.parse(parsed);
      if (Array.isArray(jsonData)) {
        return jsonData.map((item: Record<string, Object>) => this.transformToMangaInfo(item));
      }
    } catch (e) {
      logger.debug(TAG, `结果解析失败: ${e}`);
    }
  }
  return [];
}
```

```typescript
/**
 * 执行搜索（增强版）
 * 支持：关键词、URL直达、ID直达
 */
private async performSearch(): Promise<void> {
  if (!this.searchKeyword.trim()) {
    logger.warn(TAG, '搜索关键词为空');
    return;
  }

  const rawQuery = this.searchKeyword.trim();
  logger.info(TAG, `开始解析搜索查询: "${rawQuery}"`);

  // 1. 解析搜索意图
  const intent: SearchIntent = SearchIntentParser.parse(rawQuery);
  logger.info(TAG, `搜索意图: type=${intent.type}, id=${intent.id || 'N/A'}`);

  // 2. 根据意图类型分流处理
  switch (intent.type) {
    case SearchIntentType.KEYWORD:
      await this.executeKeywordSearch(rawQuery);
      break;
      
    case SearchIntentType.URL:
      await this.executeUrlDirectAccess(intent);
      break;
      
    case SearchIntentType.MANGA_ID:
      await this.executeIdDirectAccess(intent.id!, 'manga');
      break;
      
    case SearchIntentType.AUTHOR_ID:
      await this.executeAuthorSearch(intent.id!);
      break;
      
    case SearchIntentType.SERIES_ID:
      await this.executeSeriesSearch(intent.id!);
      break;
      
    case SearchIntentType.AMBIGUOUS:
      // 纯数字，需要用户选择
      this.ambiguousId = intent.id!;
      this.showAmbiguousDialog = true;
      break;
      
    default:
      await this.executeKeywordSearch(rawQuery);
  }
}

/**
 * 执行关键词搜索（原有逻辑）
 */
private async executeKeywordSearch(keyword: string): Promise<void> {
  this.showSearchDialog = false;
  this.isLoading = true;
  this.lastSearchKeyword = keyword;
  
  this.addOrUpdateSearchTab();
  this.currentTabId = 'search';
  this.searchResults = [];
  this.comicList = [];
  this.comicDataSource.updateItems([]);

  try {
    let comics: ComicInfo[] = [];
    
    if (this.useWebView && this.webViewReady && this.mangaSourceEngine) {
      logger.info(TAG, `使用WebView系统搜索: 关键词="${keyword}"`);
      const result = await this.mangaSourceEngine.searchManga(keyword, 1);
      if (result && result.success && result.data) {
        comics = this.convertMangaInfoToComicInfo(result.data.mangas);
      }
    } else {
      logger.info(TAG, `使用HTTP系统搜索: 关键词="${keyword}"`);
      comics = await this.sourceManager.search(this.sourceId, keyword, 1, 20);
    }
    
    this.searchResults = comics;
    this.comicList = comics;
    this.comicDataSource.updateItems(this.comicList);
    this.hasMore = comics.length >= 20;
  } catch (error) {
    logger.error(TAG, '执行搜索失败', String(error));
    this.getUIContext().getPromptAction().showToast({ 
      message: `搜索失败: ${error instanceof Error ? error.message : String(error)}` 
    });
  } finally {
    this.isLoading = false;
    this.isRefreshing = false;
  }
}

/**
 * 执行ID直达
 */
private async executeIdDirectAccess(id: string, type: 'manga' | 'chapter'): Promise<void> {
  this.showSearchDialog = false;
  this.isLoading = true;
  
  try {
    if (this.useWebView && this.webViewReady && this.mangaSourceEngine) {
      logger.info(TAG, `执行ID直达: id="${id}", type=${type}`);
      
      const result = await this.mangaSourceEngine.searchById(id);
      
      if (result.success && result.data && result.data.mangas.length > 0) {
        const manga = result.data.mangas[0];
        
        // 单结果直接跳转详情页
        if (result.data.mangas.length === 1) {
          logger.info(TAG, `ID直达成功，跳转详情页: ${manga.title}`);
          this.navigateToMangaDetail(manga);
          return;
        }
        
        // 多结果显示列表
        this.searchResults = this.convertMangaInfoToComicInfo(result.data.mangas);
        this.comicList = this.searchResults;
        this.comicDataSource.updateItems(this.comicList);
        this.addOrUpdateSearchTab();
        this.currentTabId = 'search';
      } else {
        this.getUIContext().getPromptAction().showToast({ 
          message: `未找到ID为 ${id} 的作品` 
        });
      }
    } else {
      // 非WebView图源，降级为关键词搜索
      logger.warn(TAG, '非WebView图源，ID直达降级为关键词搜索');
      await this.executeKeywordSearch(id);
    }
  } catch (error) {
    logger.error(TAG, 'ID直达失败', String(error));
    this.getUIContext().getPromptAction().showToast({ 
      message: `ID直达失败: ${error instanceof Error ? error.message : String(error)}` 
    });
  } finally {
    this.isLoading = false;
  }
}

/**
 * 执行作者搜索
 */
private async executeAuthorSearch(authorId: string): Promise<void> {
  this.showSearchDialog = false;
  this.isLoading = true;
  this.lastSearchKeyword = `作者: ${authorId}`;
  
  this.addOrUpdateSearchTab();
  this.currentTabId = 'search';
  this.searchResults = [];
  this.comicList = [];
  this.comicDataSource.updateItems([]);

  try {
    if (this.useWebView && this.webViewReady && this.mangaSourceEngine) {
      // 检查是否支持作者搜索
      if (this.mangaSourceEngine.supportsCapability('searchByAuthor')) {
        logger.info(TAG, `执行作者搜索: authorId="${authorId}"`);
        const result = await this.mangaSourceEngine.searchByAuthor(authorId, 1);
        
        if (result.success && result.data) {
          this.searchResults = this.convertMangaInfoToComicInfo(result.data.mangas);
          this.comicList = this.searchResults;
          this.comicDataSource.updateItems(this.comicList);
          this.hasMore = result.data.hasNext;
          return;
        }
      }
      
      // 不支持或失败，提示并降级
      logger.warn(TAG, '图源不支持作者ID搜索，降级为关键词搜索');
      this.getUIContext().getPromptAction().showToast({ 
        message: '该图源不支持作者ID搜索，已按关键词搜索' 
      });
      await this.executeKeywordSearch(authorId);
    } else {
      await this.executeKeywordSearch(authorId);
    }
  } catch (error) {
    logger.error(TAG, '作者搜索失败', String(error));
    this.getUIContext().getPromptAction().showToast({ 
      message: `作者搜索失败: ${error instanceof Error ? error.message : String(error)}` 
    });
  } finally {
    this.isLoading = false;
  }
}

/**
 * 执行URL直达
 */
private async executeUrlDirectAccess(intent: SearchIntent): Promise<void> {
  this.showSearchDialog = false;
  this.isLoading = true;
  
  try {
    const urlInfo = intent.urlParsedInfo;
    
    if (urlInfo && urlInfo.extractedId) {
      // URL已解析出ID，根据类型处理
      switch (urlInfo.pathType) {
        case 'artwork':
          await this.executeIdDirectAccess(urlInfo.extractedId, 'manga');
          return;
        case 'user':
          await this.executeAuthorSearch(urlInfo.extractedId);
          return;
        case 'series':
          await this.executeSeriesSearch(urlInfo.extractedId);
          return;
      }
    }
    
    // 无法解析，降级为关键词搜索
    logger.warn(TAG, 'URL无法解析，降级为关键词搜索');
    await this.executeKeywordSearch(intent.rawQuery);
  } catch (error) {
    logger.error(TAG, 'URL直达失败', String(error));
  } finally {
    this.isLoading = false;
  }
}

/**
 * 跳转到漫画详情页
 */
private navigateToMangaDetail(manga: MangaInfo): void {
  const params: MangaDetailPageParams = {
    sourceId: this.sourceId,
    mangaId: manga.id,
    cachedTitle: manga.title,
    cachedCover: manga.cover
  };
  this.pathStack.pushPathByName('MangaDetailPage', params);
}
```
