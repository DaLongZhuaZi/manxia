## 📝 StorageManagementPage UI实现说明

由于build方法较复杂，UI部分需要手动完成最后的集成。以下是完整的UI实现代码：

### 在build方法中的内容区域添加：

```typescript
} else if (this.statistics) {
  // 内容区域
  Scroll() {
    Column({ space: 16 }) {
      // 总览卡片
      this.buildOverviewCard()

      // 缓存详情按钮
      Button(this.showCacheDetails ? '收起缓存详情' : '查看缓存详情')
        .width('100%')
        .fontSize(14)
        .fontColor(ThemeAwareHelper.getTestManagementThemedColor('text_primary', this.themeState.currentTheme))
        .backgroundColor(ThemeAwareHelper.getTestManagementThemedColor('panel_background', this.themeState.currentTheme))
        .borderRadius(8)
        .onClick(() => {
          this.toggleCacheDetails();
        })

      // 缓存详情内容
      if (this.showCacheDetails && this.cacheDetails) {
        Column({ space: 12 }) {
          // 缓存统计
          Row() {
            Text('内存缓存')
              .fontSize(14)
              .fontColor(ThemeAwareHelper.getTestManagementThemedColor('text_secondary', this.themeState.currentTheme))
            Blank()
            Text(this.storageAnalyzer.formatSize(this.cacheDetails.memoryCacheSize))
              .fontSize(14)
              .fontWeight(FontWeight.Medium)
              .fontColor(ThemeAwareHelper.getTestManagementThemedColor('accent_primary', this.themeState.currentTheme))
          }
          .width('100%')

          Row() {
            Text('临时磁盘缓存')
              .fontSize(14)
              .fontColor(ThemeAwareHelper.getTestManagementThemedColor('text_secondary', this.themeState.currentTheme))
            Blank()
            Text(this.storageAnalyzer.formatSize(this.cacheDetails.diskCacheSize))
              .fontSize(14)
              .fontWeight(FontWeight.Medium)
              .fontColor(ThemeAwareHelper.getTestManagementThemedColor('accent_primary', this.themeState.currentTheme))
          }
          .width('100%')

          Row() {
            Text('永久磁盘缓存')
              .fontSize(14)
              .fontColor(ThemeAwareHelper.getTestManagementThemedColor('text_secondary', this.themeState.currentTheme))
            Blank()
            Text(this.storageAnalyzer.formatSize(this.cacheDetails.permanentCacheSize))
              .fontSize(14)
              .fontWeight(FontWeight.Medium)
              .fontColor(ThemeAwareHelper.getTestManagementThemedColor('accent_primary', this.themeState.currentTheme))
          }
          .width('100%')

          Divider()
            .color(ThemeAwareHelper.getTestManagementThemedColor('panel_border', this.themeState.currentTheme))

          // 漫画缓存列表
          if (this.cacheDetails.mangaCacheList.length > 0) {
            Text('漫画缓存')
              .fontSize(16)
              .fontWeight(FontWeight.Medium)
              .fontColor(ThemeAwareHelper.getTestManagementThemedColor('text_primary', this.themeState.currentTheme))

            ForEach(this.cacheDetails.mangaCacheList, (manga: MangaCacheStats) => {
              Row({ space: 12 }) {
                Column({ space: 4 }) {
                  Text(manga.mangaTitle)
                    .fontSize(14)
                    .fontWeight(FontWeight.Medium)
                    .fontColor(ThemeAwareHelper.getTestManagementThemedColor('text_primary', this.themeState.currentTheme))
                    .maxLines(1)
                    .textOverflow({ overflow: TextOverflow.Ellipsis })

                  Row({ space: 8 }) {
                    Text(this.storageAnalyzer.formatSize(manga.totalSize))
                      .fontSize(12)
                      .fontColor(ThemeAwareHelper.getTestManagementThemedColor('accent_primary', this.themeState.currentTheme))

                    Text(`${manga.imageCount} 张`)
                      .fontSize(12)
                      .fontColor(ThemeAwareHelper.getTestManagementThemedColor('text_muted', this.themeState.currentTheme))

                    Text(`${manga.chapterCount} 章`)
                      .fontSize(12)
                      .fontColor(ThemeAwareHelper.getTestManagementThemedColor('text_muted', this.themeState.currentTheme))
                  }
                }
                .layoutWeight(1)
                .alignItems(HorizontalAlign.Start)

                Button('删除')
                  .fontSize(12)
                  .fontColor(Color.White)
                  .backgroundColor(Color.Red)
                  .borderRadius(4)
                  .padding({ left: 12, right: 12, top: 6, bottom: 6 })
                  .onClick(() => {
                    this.deleteMangaCache(manga.sourceId, manga.mangaId);
                  })
              }
              .width('100%')
              .padding(12)
              .backgroundColor(ThemeAwareHelper.getTestManagementThemedColor('comp_background_secondary', this.themeState.currentTheme))
              .borderRadius(8)
            }, (manga: MangaCacheStats) => manga.mangaId)
          }

          Divider()
            .color(ThemeAwareHelper.getTestManagementThemedColor('panel_border', this.themeState.currentTheme))

          // 封面缓存列表
          if (this.cacheDetails.coverCacheList.length > 0) {
            Text('封面缓存')
              .fontSize(16)
              .fontWeight(FontWeight.Medium)
              .fontColor(ThemeAwareHelper.getTestManagementThemedColor('text_primary', this.themeState.currentTheme))

            ForEach(this.cacheDetails.coverCacheList, (cover: SourceCoverInfo) => {
              Row({ space: 12 }) {
                Column({ space: 4 }) {
                  Text(cover.sourceName)
                    .fontSize(14)
                    .fontWeight(FontWeight.Medium)
                    .fontColor(ThemeAwareHelper.getTestManagementThemedColor('text_primary', this.themeState.currentTheme))

                  Row({ space: 8 }) {
                    Text(this.storageAnalyzer.formatSize(cover.totalSize))
                      .fontSize(12)
                      .fontColor(ThemeAwareHelper.getTestManagementThemedColor('accent_primary', this.themeState.currentTheme))

                    Text(`${cover.coverCount} 个封面`)
                      .fontSize(12)
                      .fontColor(ThemeAwareHelper.getTestManagementThemedColor('text_muted', this.themeState.currentTheme))
                  }
                }
                .layoutWeight(1)
                .alignItems(HorizontalAlign.Start)

                Button('删除')
                  .fontSize(12)
                  .fontColor(Color.White)
                  .backgroundColor(Color.Red)
                  .borderRadius(4)
                  .padding({ left: 12, right: 12, top: 6, bottom: 6 })
                  .onClick(() => {
                    this.deleteSourceCovers(cover.sourceId);
                  })
              }
              .width('100%')
              .padding(12)
              .backgroundColor(ThemeAwareHelper.getTestManagementThemedColor('comp_background_secondary', this.themeState.currentTheme))
              .borderRadius(8)
            }, (cover: SourceCoverInfo) => cover.sourceId.toString())
          }
        }
        .width('100%')
        .padding(16)
        .backgroundColor(ThemeAwareHelper.getTestManagementThemedColor('panel_background', this.themeState.currentTheme))
        .borderRadius(12)
      }

      // 存储项列表
      ForEach(this.statistics.items, (item: StorageItem) => {
        this.buildStorageItemCard(item)
      }, (item: StorageItem) => item.type)
    }
    .padding({ left: 16, right: 16, top: 8, bottom: 80 })
  }
  .layoutWeight(1)
  .scrollBar(BarState.Auto)
  .edgeEffect(EdgeEffect.Spring)

  // 操作按钮
  this.buildActionButtons()
}
```
