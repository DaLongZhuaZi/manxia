```typescript
// processCoverCache方法
private async processCoverCache(comics: ComicInfo[]): Promise<void> {
  const coverCache = CoverCacheManager.getInstance();
  const headers: Record<string, string> = { /* 转换后的headers */ };

  for (const comic of comics) {
    // 保存到temp_manga_info表
    await coverCache.saveTempManga(
      this.sourceId,
      comic.id,
      comic.title,
      comic.coverUrl
    );

    // 检查缓存
    const localPath = await coverCache.getLocalCoverPath(
      this.sourceId,
      comic.id
    );

    if (!localPath) {
      // 后台下载
      coverCache.downloadAndCacheCover(
        this.sourceId,
        comic.id,
        comic.coverUrl,
        headers
      ).catch(() => {});
    }
  }
}
```

**数据流程**:
```
用户访问图源详情页
    ↓
loadComics() 加载漫画列表
    ↓
processCoverCache() 处理封面缓存
    ↓
saveTempManga() → 保存到temp_manga_info表
    ↓
getLocalCoverPath() → 检查本地缓存
    ↓
有缓存 → 记录日志
无缓存 → downloadAndCacheCover() 后台下载
    ↓
下载完成 → 更新localCoverPath字段
```

**实现位置**: `ImageCacheManager.performImageLoad()`

```typescript
import { PermanentImageCacheManager } from './PermanentImageCacheManager';

private async performImageLoad(
  url: string,
  cacheKey: string,
  options?: LoadOptions
): Promise<image.PixelMap | null> {
  try {
    // 第一级：内存缓存
    if (!options?.skipMemoryCache && !options?.forceRefresh) {
      const memoryResult = this.getFromMemoryCache(cacheKey);
      if (memoryResult) {
        return memoryResult;
      }
    }
    
    // 第二级：磁盘缓存（临时）
    if (!options?.skipDiskCache && !options?.forceRefresh) {
      const diskResult = await this.getFromDiskCache(cacheKey, options);
      if (diskResult) {
        if (!options?.skipMemoryCache) {
          await this.putToMemoryCache(cacheKey, diskResult);
        }
        return diskResult;
      }
    }
    
    // 新增：第三级：永久缓存（如果启用了downloadWhileReading）
    if (options?.sourceId && options?.mangaId && options?.chapterId && options?.pageIndex !== undefined) {
      const permanentCache = PermanentImageCacheManager.getInstance();
      const cachedPath = await permanentCache.getCachedImagePath(
        options.sourceId,
        options.mangaId,
        options.chapterId,
        options.pageIndex,
        url
      );
      
      if (cachedPath) {
        logger.debug(TAG, `永久缓存命中: ${url}`);
        const pixelMap = await this.loadImageFromFile(cachedPath);
        if (pixelMap) {
          // 存入内存缓存
          if (!options?.skipMemoryCache) {
            await this.putToMemoryCache(cacheKey, pixelMap);
          }
          return pixelMap;
        }
      }
    }
    
    // 第四级：网络加载
    logger.debug(TAG, `网络加载: ${url}`);
    const networkResult = await this.loadFromNetwork(url, options);
    
    if (networkResult) {
      // 保存到永久缓存（如果启用）
      if (options?.sourceId && options?.mangaId && options?.chapterId && options?.pageIndex !== undefined) {
        const permanentCache = PermanentImageCacheManager.getInstance();
        
        // 将PixelMap转换为ArrayBuffer
        const imageData = await this.pixelMapToArrayBuffer(networkResult);
        
        await permanentCache.saveImageToCache(
          options.sourceId,
          options.mangaId,
          options.chapterId,
          options.pageIndex,
          url,
          imageData
        );
      }
      
      // 存入临时磁盘缓存
      if (!options?.skipDiskCache) {
        await this.putToDiskCache(cacheKey, networkResult, options);
      }
      
      // 存入内存缓存
      if (!options?.skipMemoryCache) {
        await this.putToMemoryCache(cacheKey, networkResult);
      }
    }
    
    return networkResult;
    
  } catch (error) {
    logger.error(TAG, `执行图片加载失败: ${url}` + String(error));
    throw error as Error;
  }
}

// 辅助方法：PixelMap转ArrayBuffer
private async pixelMapToArrayBuffer(pixelMap: image.PixelMap): Promise<ArrayBuffer> {
  const imagePackerApi = image.createImagePacker();
  const packOpts: image.PackingOption = { format: 'image/jpeg', quality: 95 };
  const data = await imagePackerApi.packing(pixelMap, packOpts);
  return data;
}

// 辅助方法：从文件加载PixelMap
private async loadImageFromFile(filePath: string): Promise<image.PixelMap | null> {
  try {
    const imageSourceApi = image.createImageSource(filePath);
    const pixelMap = await imageSourceApi.createPixelMap();
    return pixelMap;
  } catch (error) {
    logger.error(TAG, `从文件加载图片失败: ${filePath}`, String(error));
    return null;
  }
}
```

3. **错误处理**: 所有缓存操作都有try-catch保护
4. **日志记录**: 详细记录缓存操作便于调试
5. **性能优化**: 后台下载不阻塞UI渲染
