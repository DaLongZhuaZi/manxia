### 2. 添加缺失的清理方法

#### PermanentImageCacheManager

需要添加`cleanMangaCache`方法：

```typescript
/**
 * 清理指定漫画的缓存
 */
async cleanMangaCache(sourceId: number, mangaId: string): Promise<void> {
  const mangaPath = `${this.permanentCachePath}/${sourceId}/${mangaId}`;
  
  try {
    if (fs.accessSync(mangaPath)) {
      fs.rmdirSync(mangaPath);
      logger.info(TAG, `已删除漫画缓存: ${mangaPath}`);
    }
  } catch (error) {
    logger.error(TAG, `删除漫画缓存失败: ${error}`);
    throw error;
  }
}
```

#### CoverCacheManager

需要添加`cleanSourceCovers`方法：

```typescript
/**
 * 清理指定图源的所有封面
 */
async cleanSourceCovers(sourceId: number): Promise<void> {
  const sourcePath = `${this.coverCachePath}/${sourceId}`;
  
  try {
    if (fs.accessSync(sourcePath)) {
      fs.rmdirSync(sourcePath);
      logger.info(TAG, `已删除图源封面: ${sourcePath}`);
      
      // 同时清理数据库记录
      const dbManager = DatabaseManager.getInstance();
      await dbManager.execute(
        'DELETE FROM temp_manga_info WHERE sourceId = ?',
        [sourceId]
      );
    }
  } catch (error) {
    logger.error(TAG, `删除图源封面失败: ${error}`);
    throw error;
  }
}
```

@Builder

            .fontSize(14)
            .fontColor(ThemeAwareHelper.getTestManagementThemedColor('text_secondary', this.themeState.currentTheme))
          Blank()
          Text(this.storageAnalyzer.formatSize(this.cacheDetails.diskCacheSize))
            .fontSize(14)
            .fontWeight(FontWeight.Medium)
            .fontColor(ThemeAwareHelper.getTestManagementThemedColor('accent_primary', this.themeState.currentTheme))
        }
        .width('100%')

        // 永久磁盘缓存
        Row() {
          Text('永久磁盘缓存')
            .fontSize(14)
            .fontColor(ThemeAwareHelper.getTestManagementThemedColor('text_secondary', this.themeState.currentTheme))
          Blank()
          Text(this.storageAnalyzer.formatSize(this.cacheDetails.permanentCacheSize))
            .fontSize(14)
            .fontWeight(FontWeight.Medium)

        }

          .fontSize(16)

          this.buildCoverCacheItem(cover)
        }, (cover: SourceCoverInfo) => cover.sourceId.toString())
      }
    }
  }
  .width('100%')
  .padding(16)
  .backgroundColor(ThemeAwareHelper.getTestManagementThemedColor('panel_background', this.themeState.currentTheme))
  .borderRadius(12)
